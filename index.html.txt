                    window.lucide.createIcons();

                    if (window.initFirebase) {

                            setDbInstance(db);
                            setIsFirebaseConnected(true);
                            
                            const leaveColRef = collection(db, 'artifacts', window.appId, 'public', 'data', 'leave_requests');
                            
                            unsubscribe = onSnapshot(leaveColRef, (snapshot) => {
                                if (!snapshot.empty) {
                                    const fetchedLeaves = [];
                                    snapshot.forEach((doc) => {
                                        fetchedLeaves.push({ id: doc.id, ...doc.data() });
                                    });
                                    fetchedLeaves.sort((a, b) => new Date(b.createdAt) - new Date(a.createdAt));
                                    setLeaves(fetchedLeaves);
                                }
                            }, (error) => {
                                console.error("Firestore sync error:", error);
                            });
                        }
                    }
                }
                setupFirebase();
                return () => unsubscribe();

            };

                setStaffList(prev => prev.filter(s => s.id !== staffId));

                    setCurrentUser(null);

                    try {
                        const leaveDocRef = doc(dbInstance, 'artifacts', window.appId, 'public', 'data', 'leave_requests', leaveId);
                        await window.deleteDoc(leaveDocRef);

                        console.error("Error deleting document from Firestore:", err);
                    }
                }
                showNotification('ลบรายการใบลาเรียบร้อยแล้ว');
            };

                            ...item,
                            status: newStatus,
                            approverComment: comment,
                            approverName: approverName,
                            approvalDate: new Date().toISOString().split('T')[0]
                        };
                    }
                    return item;
                });

                            status: newStatus,
                            approverComment: comment,
                            approverName: approverName,
                            approvalDate: new Date().toISOString().split('T')[0]

                    } catch (err) {
                        console.error("Error updating status in Firestore:", err);
                    }
                }

                showNotification(`บันทึกการ${newStatus === 'approved' ? 'อนุญาต' : 'ไม่อนุญาต'}เรียบร้อยแล้ว`);

                                                    <i data-lucide="shield-check" className="w-3.5 h-3.5 mr-1"></i>

                                        <div className="flex items-center gap-3 bg-white/10 p-2 rounded-xl border border-white/10">

                                                </div>

                                                className="p-1.5 text-teal-200 hover:text-white hover:bg-white/10 rounded-lg transition-all"
                                                title="ออกจากระบบบุคลากร"
                                            >
                                                <i data-lucide="log-out" className="w-4 h-4"></i>
                                            </button>
                                        </div>

                                        <button
                                            onClick={() => setShowStaffLoginModal(true)}
                                            className="px-3.5 py-2 bg-emerald-600 hover:bg-emerald-500 text-white rounded-xl text-xs font-bold shadow transition-all flex items-center gap-1.5 active:scale-95 border border-emerald-400/30"

            const [password, setPassword] = useState('123');
            const [loginMode, setLoginMode] = useState('quick'); // 'quick' or 'manual'

            // Auto sync username/password when select staff changed in quick mode
            useEffect(() => {
                const found = staffList.find(s => s.id === selectedStaffId);
                if (found) {
                    setUsername(found.username || found.id);
                    setPassword(found.password || '123');
                }
            }, [selectedStaffId, staffList]);

            const handleSubmit = (e) => {
                e.preventDefault();
                if (loginMode === 'quick') {
                    const found = staffList.find(s => s.id === selectedStaffId);
                    if (found) {
                        onLogin('', '', found);
                        return;
